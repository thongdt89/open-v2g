# Separate candidate qualification target. No ISO-20 feature gate or vehicle authority is enabled.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/SHA256SUMS" _iso20_checksums)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/SHA256SUMS")
set(_iso20_sources)
foreach(_line IN LISTS _iso20_checksums)
    string(SUBSTRING "${_line}" 0 64 _expected)
    string(SUBSTRING "${_line}" 66 -1 _relative)
    set(_file "${OPENV2G_CBV2G_SOURCE_DIR}/${_relative}")
    if(NOT EXISTS "${_file}")
        message(FATAL_ERROR "Missing pinned ISO-20 artifact: ${_relative}")
    endif()
    file(SHA256 "${_file}" _actual)
    if(NOT _actual STREQUAL _expected)
        message(FATAL_ERROR "Pinned ISO-20 artifact checksum mismatch: ${_relative}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_file}")
    if(_relative MATCHES "\\.c$")
        list(APPEND _iso20_sources "${_file}")
    endif()
endforeach()
add_library(openv2g_cbv2g_iso20_generated STATIC ${_iso20_sources})
target_include_directories(openv2g_cbv2g_iso20_generated PRIVATE "${OPENV2G_CBV2G_SOURCE_DIR}/include")
target_compile_features(openv2g_cbv2g_iso20_generated PRIVATE c_std_11)
target_link_libraries(openv2g_cbv2g_iso20_generated PRIVATE openv2g_cbv2g_sap_generated)
if(NOT MSVC)
    target_compile_options(openv2g_cbv2g_iso20_generated PRIVATE -Wall -Wextra -Wpedantic)
endif()

    target_link_libraries(openv2g_test_iso20_models PRIVATE openv2g_cbv2g_iso20_generated openv2g_cbv2g_sap_generated)
    target_include_directories(openv2g_test_iso20_models PRIVATE "${OPENV2G_CBV2G_SOURCE_DIR}/include" "${PROJECT_SOURCE_DIR}/tests")
    openv2g_enable_strict_warnings(openv2g_test_iso20_models)
