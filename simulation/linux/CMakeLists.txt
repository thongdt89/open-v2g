option(OPENV2G_ENABLE_OPENSSL_HOST "Build real OpenSSL host TLS/crypto provider" OFF)
if(OPENV2G_ENABLE_OPENSSL_HOST)
    find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
    add_library(openv2g_openssl_host STATIC src/tls.c src/crypto.c)
    target_link_libraries(openv2g_openssl_host PUBLIC openv2g_linux_bsw PRIVATE OpenSSL::SSL OpenSSL::Crypto)
    target_include_directories(openv2g_openssl_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    # Also support a distro development package extracted outside /usr.
    if(EXISTS "${OPENSSL_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}/openssl/opensslconf.h")
        target_include_directories(openv2g_openssl_host PRIVATE "${OPENSSL_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}")
    endif()
    target_compile_definitions(openv2g_openssl_host PUBLIC OPENV2G_HAVE_OPENSSL_HOST=1)
    openv2g_enable_strict_warnings(openv2g_openssl_host)
endif()
if(OPENV2G_ENABLE_OPENSSL_HOST AND OPENV2G_BUILD_TESTS)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import cryptography" RESULT_VARIABLE _crypto_python)
    if(NOT _crypto_python EQUAL 0)
        message(FATAL_ERROR "Host TLS tests require Python cryptography for ephemeral certificate fixtures")
    endif()
    add_executable(openv2g_test_linux_tls tests/test_tls.c)
    target_link_libraries(openv2g_test_linux_tls PRIVATE openv2g_openssl_host)
    target_include_directories(openv2g_test_linux_tls PRIVATE ${PROJECT_SOURCE_DIR}/tests)
    openv2g_enable_strict_warnings(openv2g_test_linux_tls)
    add_test(NAME openv2g_test_linux_tls COMMAND ${Python3_EXECUTABLE}
        ${PROJECT_SOURCE_DIR}/tests/security/fixture.py $<TARGET_FILE:openv2g_test_linux_tls>)
    set_tests_properties(openv2g_test_linux_tls PROPERTIES TIMEOUT 30)
endif()
